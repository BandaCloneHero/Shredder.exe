// Com o portal aberto e conectado, cole este trecho no Console (F12) e pressione Enter.
// A tabela mostra nomes de chaves e se o valor contém 1012. Não mostra valores nem tokens.
(() => {
  const id = "1012";
  const linhas = [];

  for (const armazenamento of ["localStorage", "sessionStorage"]) {
    const dados = window[armazenamento];
    for (let i = 0; i < dados.length; i++) {
      const chave = dados.key(i);
      const valor = dados.getItem(chave) || "";
      if (/user|player|jogador|conta|id/i.test(chave) || valor.includes(id)) {
        linhas.push({ armazenamento, chave, contem_1012: valor.includes(id) });
      }
    }
  }

  if (linhas.length) console.table(linhas);
  else console.log("Nenhuma chave correspondente encontrada neste armazenamento.");
})();
